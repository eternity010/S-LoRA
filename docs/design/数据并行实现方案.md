使用 **ZMQ (ZeroMQ)** 进行进程间通信：

```python
# Router -> Workers: PUSH/PULL 模式

               ──────> Worker 2 (PULL)
               ──────> Worker 3 (PULL)

Worker 1 (PUB) ──────┤
Worker 2 (PUB) ──────┼──> Router (SUB)
Worker 3 (PUB) ──────┘
```

## 三、核心组件实现

### 3.1 Data Parallel Router Manager

**文件**: `slora/server/router/dp_manager.py`

**核心职责**：
- 启动和管理多个 GPU Worker 进程
- 接收来自 API Server 的请求
- 根据路由策略分发请求到 Worker
- 收集 Worker 状态信息
- 处理 Worker 故障和重启

**关键代码结构**：

```python
class DataParallelRouterManager:
    def __init__(self, args, num_workers):
        self.num_workers = num_workers
        self.workers = []
        self.worker_states = {}  # 存储每个 worker 的状态
        
        # 初始化路由策略
        self.router = self._create_router(args.routing_strategy)
        
        # ZMQ 通信
        self.context = zmq.asyncio.Context()
        self.request_sender = self.context.socket(zmq.PUSH)
        self.response_receiver = self.context.socket(zmq.PULL)
        self.state_subscriber = self.context.socket(zmq.SUB)
    
    async def start_workers(self):
        """启动所有 GPU Worker 进程"""
        for worker_id in range(self.num_workers):
            worker = await self._start_worker(worker_id)
            self.workers.append(worker)
    
    async def route_request(self, request):
        """路由请求到合适的 Worker"""
        worker_id = self.router.select_worker(
            request, 
            self.worker_states
        )
        await self._send_to_worker(worker_id, request)
    
    async def collect_worker_states(self):
        """收集 Worker 状态信息"""
        while True:
            state_msg = await self.state_subscriber.recv_json()
            self.worker_states[state_msg['worker_id']] = state_msg
            await asyncio.sleep(0.1)
```

```python
class GPUWorker:
    def __init__(self, worker_id, gpu_id, args):
        self.worker_id = worker_id
        self.gpu_id = gpu_id
        
        # 设置 GPU
        os.environ['CUDA_VISIBLE_DEVICES'] = str(gpu_id)
        torch.cuda.set_device(0)  # 在进程内是 0
        
        # 加载模型（复用现有代码）
        self.model = self._load_model(args)
        
        # Adapter 缓存
        self.adapter_cache = AdapterCache(
            max_size=args.adapter_cache_size
        )
        
        # 请求队列
        self.request_queue = asyncio.Queue()
        
        # ZMQ 通信
        self.context = zmq.asyncio.Context()
        self.request_receiver = self.context.socket(zmq.PULL)
        self.response_sender = self.context.socket(zmq.PUSH)
        self.state_publisher = self.context.socket(zmq.PUB)
    
    async def run(self):
        """主循环"""
        # 启动状态上报任务
        asyncio.create_task(self._report_state_loop())
        
        # 启动请求处理任务
        asyncio.create_task(self._receive_requests_loop())
        asyncio.create_task(self._process_requests_loop())
        
        await asyncio.Event().wait()  # 永久运行
    
    async def _process_requests_loop(self):
        """处理请求队列"""
        while True:
            request = await self.request_queue.get()
            
            try:
                # 确保 adapter 已加载
                if request.adapter_dir not in self.adapter_cache:
                    await self._load_adapter(request.adapter_dir)
                
                # 执行推理（复用现有逻辑）
                result = await self._infer(request)
                
                # 发送响应
                await self.response_sender.send_json({
                    'request_id': request.request_id,
                    'result': result,
                    'worker_id': self.worker_id
                })
            except Exception as e:
                # 错误处理
                await self._send_error(request.request_id, str(e))
    
    async def _report_state_loop(self):
        """定期上报状态"""
        while True:
            state = {
                'worker_id': self.worker_id,
                'queue_length': self.request_queue.qsize(),
                'cached_adapters': list(self.adapter_cache.keys()),
                'gpu_memory_used': torch.cuda.memory_allocated(),
                'timestamp': time.time()
            }
            await self.state_publisher.send_json(state)
            await asyncio.sleep(1.0)  # 每秒上报一次
```

### 3.3 Request Router（路由策略）

**文件**: `slora/server/router/request_router.py`

**三种路由策略实现**：

```python
class RequestRouter(ABC):
    @abstractmethod
    def select_worker(self, request, worker_states):
        """选择目标 Worker"""
        pass

# 策略 1: 轮询
class RoundRobinRouter(RequestRouter):
    def __init__(self, num_workers):
        self.num_workers = num_workers
        self.counter = 0
    
    def select_worker(self, request, worker_states):
        worker_id = self.counter % self.num_workers
        self.counter += 1
        return worker_id

# 策略 2: 最少连接
class LeastConnectionRouter(RequestRouter):
    def select_worker(self, request, worker_states):
        # 选择队列最短的 Worker
        min_queue = float('inf')
        selected_worker = 0
        
        for worker_id, state in worker_states.items():
            if state['queue_length'] < min_queue:
                min_queue = state['queue_length']
                selected_worker = worker_id
        
        return selected_worker

# 策略 3: Adapter 亲和性（推荐）
class AffinityRouter(RequestRouter):
    def __init__(self, cache_weight=0.7, load_weight=0.3):
        self.cache_weight = cache_weight
        self.load_weight = load_weight
    
    def select_worker(self, request, worker_states):
        adapter = request.adapter_dir
        scores = {}
        
        for worker_id, state in worker_states.items():
            # 计算缓存得分
            cache_score = 1.0 if adapter in state['cached_adapters'] else 0.0
            
            # 计算负载得分（队列越短得分越高）
            max_queue = max(s['queue_length'] for s in worker_states.values())
            load_score = 1.0 - (state['queue_length'] / (max_queue + 1))
            
            # 综合得分
            scores[worker_id] = (
                self.cache_weight * cache_score +
                self.load_weight * load_score
            )
        
        # 选择得分最高的 Worker
        return max(scores, key=scores.get)
```

### 3.4 Adapter Cache Manager

**文件**: `slora/server/router/adapter_cache.py`

**LRU 缓存实现**：

```python
from collections import OrderedDict
from typing import Optional

class AdapterCache:
    def __init__(self, max_size: int = 100):
        self.cache = OrderedDict()
        self.max_size = max_size
        self.access_count = {}
        self.hit_count = 0
        self.miss_count = 0
    
    def get(self, adapter_dir: str) -> Optional[object]:
        """获取 adapter，如果存在则移到最后（最近使用）"""
        if adapter_dir in self.cache:
            self.cache.move_to_end(adapter_dir)
            self.access_count[adapter_dir] = self.access_count.get(adapter_dir, 0) + 1
            self.hit_count += 1
            return self.cache[adapter_dir]
        
        self.miss_count += 1
        return None
    
    def put(self, adapter_dir: str, adapter: object):
        """添加 adapter 到缓存"""
        if adapter_dir in self.cache:
            # 已存在，移到最后
            self.cache.move_to_end(adapter_dir)
        else:
            # 新增
            if len(self.cache) >= self.max_size:
                # 缓存已满，淘汰最久未使用的
                evicted_key, _ = self.cache.popitem(last=False)
                print(f"[Cache] Evicted adapter: {evicted_key}")
            
            self.cache[adapter_dir] = adapter
            self.access_count[adapter_dir] = 1
    
    def keys(self):
        """返回所有缓存的 adapter"""
        return list(self.cache.keys())
    
    def __contains__(self, adapter_dir: str):
        return adapter_dir in self.cache
    
    def get_hit_rate(self):
        """计算缓存命中率"""
        total = self.hit_count + self.miss_count
        return self.hit_count / total if total > 0 else 0.0
```

## 九、常见问题

### Q1: 如何选择合适的 Worker 数量？

**A**: 通常设置为 GPU 数量。如果单个模型很小，可以在一张 GPU 上运行多个 Worker。

```python
# 示例：2 张 GPU，每张运行 2 个 Worker
--num-workers 4 --gpu-ids 0,0,1,1
```

### Q2: 数据并行和张量并行如何选择？

**A**: 
- **数据并行**: 模型能放进单卡 + 高并发场景
- **张量并行**: 模型太大单卡放不下 + 低延迟要求

### Q3: 缓存大小如何设置？

**A**: 根据 adapter 数量和 GPU 显存：
- Adapter 总数 < 100: cache_size = 50-100
- Adapter 总数 100-500: cache_size = 100-200
- Adapter 总数 > 500: cache_size = 200-500

### Q4: 如何处理 Worker 崩溃？

**A**: 系统会自动检测和重启。如果频繁崩溃，检查：
- GPU 显存是否足够
- 模型加载是否正确
- 日志中的错误信息

### Q5: 性能不达预期怎么办？

**A**: 检查以下方面：
1. GPU 利用率是否 > 90%
2. 缓存命中率是否 > 70%
3. 负载是否均衡
4. 网络延迟是否正常
